==================
Fallible traversal is a while over tried steps, and a labelled loop
==================

visit: () -> () ! io_failed =
    mut cursor := try first(source)
    while not (try at_end(source, cursor)) do
        use(try item(source, cursor))
        cursor = try next(source, cursor)
    end while
    scan: for value in source do
        break scan when value
    end scan
end visit
